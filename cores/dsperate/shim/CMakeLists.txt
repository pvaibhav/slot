cmake_minimum_required(VERSION 3.16)
project(slot_ds LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DSPERATE_SRC "" CACHE PATH "DSperate source tree")
add_subdirectory(${DSPERATE_SRC} dsperate)
add_library(slot_ds SHARED slot_ds.cpp blip_buf/blip_buf.c)
target_link_libraries(slot_ds PRIVATE dsperate_core)
if(NOT APPLE)
  target_link_options(slot_ds PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL -Wl,-z,defs)
endif()
set_target_properties(slot_ds PROPERTIES PREFIX "" OUTPUT_NAME slot_ds CXX_VISIBILITY_PRESET default)
